cmake_minimum_required(VERSION 3.24)
project(FreeClimb VERSION 0.2.30 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(FREECLIMB_PLUGIN "Build the Windows SKSE plugin" ON)
option(BUILD_TESTING "Build regression tests" OFF)
option(FREECLIMB_LOCAL_TESTS "Enable checks using this developer's installed game and tools" OFF)
option(FREECLIMB_AUTHORING_TOOLS "Build local motion authoring tools" OFF)
set(FREECLIMB_MOTION_FILE "${CMAKE_SOURCE_DIR}/runtime/meshes/actors/character/animations/FreeClimb/pack.json" CACHE FILEPATH "Optional animation pack for asset tests")
set(FREECLIMB_LEGACY_MOTION_FILE "" CACHE FILEPATH "Optional legacy binary for offline migration compatibility tests")
set(FREECLIMB_HKX_DIRECTORY "${CMAKE_SOURCE_DIR}/runtime/meshes/actors/character/animations/FreeClimb" CACHE PATH "Optional HKX directory for equivalence tests")
set(FREECLIMB_GAME_DIR "" CACHE PATH "Installed Skyrim directory for local ABI checks")
set(FREECLIMB_MO2_DIR "" CACHE PATH "MO2 directory for local skeleton checks")
set(FREECLIMB_MO2_PROFILE "" CACHE STRING "MO2 profile for local skeleton checks")

add_library(FreeClimbAnimationInput STATIC src/HkxAnimation.cpp src/HkxSpline.cpp src/AnimationOverrides.cpp src/AnimationPack.cpp)
target_include_directories(FreeClimbAnimationInput PUBLIC src external/nlohmann)
add_library(FreeClimbSettings STATIC src/UserSettings.cpp src/TranslationCatalog.cpp)
target_include_directories(FreeClimbSettings PUBLIC src)
if(MSVC)
  target_compile_options(FreeClimbAnimationInput PRIVATE /utf-8 /W4)
  target_compile_options(FreeClimbSettings PRIVATE /utf-8 /W4)
endif()
if(FREECLIMB_AUTHORING_TOOLS)
  add_executable(FreeClimbAuthoring tools/AuthoringCli.cpp)
  target_link_libraries(FreeClimbAuthoring PRIVATE FreeClimbAnimationInput)
  add_executable(FreeClimbMigrateAnimationPack tools/MigrateAnimationPack.cpp)
  target_link_libraries(FreeClimbMigrateAnimationPack PRIVATE FreeClimbAnimationInput)
  add_executable(FreeClimbExportMotionHkx tools/ExportMotionHkx.cpp)
  target_include_directories(FreeClimbExportMotionHkx PRIVATE src)
  target_link_libraries(FreeClimbExportMotionHkx PRIVATE FreeClimbAnimationInput)
  if(MSVC)
    target_compile_options(FreeClimbAuthoring PRIVATE /utf-8 /W4)
    target_compile_options(FreeClimbMigrateAnimationPack PRIVATE /utf-8)
    target_compile_options(FreeClimbExportMotionHkx PRIVATE /utf-8)
  endif()
endif()

if(BUILD_TESTING)
  enable_testing()
  include(cmake/Tests.cmake)
endif()

if(FREECLIMB_PLUGIN)
  set(ENABLE_SKYRIM_SE ON CACHE BOOL "" FORCE)
  set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
  set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
  set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(COMMONLIB_PREBUILT OFF CACHE BOOL "" FORCE)
  set(COMMONLIB_PREBUILT_DIR "" CACHE PATH "" FORCE)
  set(COMMONLIB_PREBUILT_MULTICONFIG OFF CACHE BOOL "" FORCE)
  set(COMMONLIB_ENABLE_IPO OFF CACHE BOOL "")
  set(SKSE_SUPPORT_PATCH_SAFETY ON CACHE BOOL "" FORCE)
  set(FETCHCONTENT_SOURCE_DIR_HDE64 "${CMAKE_SOURCE_DIR}/external/MinHook" CACHE PATH "" FORCE)
  list(PREPEND CMAKE_PREFIX_PATH "${CMAKE_SOURCE_DIR}/external/install-multiruntime")
  set(RAPIDCSV_INCLUDE_DIRS "${CMAKE_SOURCE_DIR}/external/rapidcsv/src" CACHE PATH "")
  find_package(spdlog CONFIG REQUIRED)
  add_subdirectory(external/CommonLibNG)
  if(MSVC)
    target_compile_options(CommonLibSSE PRIVATE /utf-8)
  endif()
  include(external/CommonLibNG/cmake/CommonLibSSE.cmake)
  add_commonlibsse_plugin(FreeClimb AUTHOR "Epsilona"
    COMPATIBLE_RUNTIMES 1.5.97.0 1.6.317.0 1.6.318.0 1.6.323.0 1.6.342.0 1.6.353.0 1.6.629.0 1.6.640.0 1.6.659.1 1.6.1130.0 1.6.1170.0 1.6.1179.1 1.7.99.0 1.7.104.0
    SOURCES src/Plugin.cpp src/SettingsMenu.cpp)
  target_include_directories(FreeClimb PRIVATE src)
  target_link_libraries(FreeClimb PRIVATE FreeClimbAnimationInput FreeClimbSettings)
  target_compile_options(FreeClimb PRIVATE /utf-8 /W4 /Zi)
  target_link_options(FreeClimb PRIVATE /DEBUG:FULL /OPT:REF /OPT:ICF /PDBALTPATH:%_PDB% /MAP)
  target_precompile_headers(FreeClimb PRIVATE src/PCH.h)
  install(TARGETS FreeClimb RUNTIME DESTINATION SKSE/Plugins)
  if(BUILD_TESTING)
    add_executable(FreeClimbRuntimeLogTests tests/RuntimeLogTests.cpp)
    target_include_directories(FreeClimbRuntimeLogTests PRIVATE src)
    target_link_libraries(FreeClimbRuntimeLogTests PRIVATE spdlog::spdlog)
    target_compile_options(FreeClimbRuntimeLogTests PRIVATE /utf-8)
    add_test(NAME runtime_log COMMAND FreeClimbRuntimeLogTests "${CMAKE_CURRENT_BINARY_DIR}/runtime-log-tests")
    add_executable(FreeClimbFilteredRayCollectorTests tests/FilteredRayCollectorTests.cpp)
    target_include_directories(FreeClimbFilteredRayCollectorTests PRIVATE src)
    target_link_libraries(FreeClimbFilteredRayCollectorTests PRIVATE CommonLibSSE::CommonLibSSE)
    target_compile_options(FreeClimbFilteredRayCollectorTests PRIVATE /utf-8)
    add_test(NAME filtered_ray_collector COMMAND FreeClimbFilteredRayCollectorTests)
    add_executable(FreeClimbRuntimeBridgeTests
      tests/RuntimeBridgeTests.cpp
      external/CommonLibNG/src/REL/Module.cpp
      external/CommonLibNG/src/REL/IDDB.cpp
      external/CommonLibNG/src/REL/Version.cpp
      external/CommonLibNG/src/REX/W32.cpp)
    target_include_directories(FreeClimbRuntimeBridgeTests PRIVATE
      src external/CommonLibNG/include external/CommonLibNG/extern/openvr/headers
      external/install-multiruntime/include external/DirectXTK/Inc external/rapidcsv/src)
    target_compile_definitions(FreeClimbRuntimeBridgeTests PRIVATE
      ENABLE_COMMONLIBSSE_TESTING=1 ENABLE_SKYRIM_SE=1 ENABLE_SKYRIM_AE=1
      HAS_SKYRIM_MULTI_TARGETING=1 SPDLOG_COMPILED_LIB WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(FreeClimbRuntimeBridgeTests PRIVATE /FISKSE/Impl/PCH.h /Zc:preprocessor /utf-8)
    target_link_libraries(FreeClimbRuntimeBridgeTests PRIVATE spdlog::spdlog
      advapi32.lib bcrypt.lib d3d11.lib d3dcompiler.lib dbghelp.lib dxgi.lib
      ole32.lib version.lib shell32.lib user32.lib)
    add_test(NAME runtime_bridge COMMAND FreeClimbRuntimeBridgeTests)
  endif()
endif()
